## 1. 시계열 데이터(Time Series Data) 
- 시간의 흐름에 따른 연속적인 데이터
- 시각(점,순간) vs 시간(선분,간격,구간)
- 사용이유: 미래예측, 불확실성 감소를 위해서
- 예제)
~~~
import seaborn as sns
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# 난수 생성 시드 고정
np.random.seed(42)
# 2023-01-01부터 100일간 날짜 생성
date_range = pd.date_range(start="2023-01-01", periods=100, freq="D")
# 평균이 0이고 표준편차가 1인 정규분포 난수 100개 생성, cumsum()은 배열 누적합 계산:배열의 각 요소를 앞에서부터 차례대로 더한 값을 새로운 배열로 반환
data = np.cumsum(np.random.randn(100))

df = pd.DataFrame({"Date": date_range, "Value": data})

# 그래프 크기 설정
plt.figure(figsize=(10,5))
# 시계열 선 그래프(색상, 마커 모양 설정)
sns.lineplot(x="Date", y="Value", data=df, color="blue", marker="o") 
plt.xlabel("Date")  # X축 라벨 설정
plt.ylabel("Value")  # Y축 라벨 설정
plt.title("Time Series Line Plot")
# X축 눈금 45도 회전(레이블이 길거나 간격이 좁을때 서로 겹치는 경우를 해결)
plt.xticks(rotation=45)

plt.show() 
~~~
